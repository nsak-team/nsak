\begin{frame}{Containerlab -- Qualitative Results}
    \framesubtitle{The discriminator: extracting the LDAP cleartext credentials}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.54\textwidth}
            \scriptsize
            \begin{itemize}
                \item \textbf{claude-opus-4-7} extracts the credentials in \textbf{every run} + correct attack-chain reasoning (reuse $\rightarrow$ lateral movement). Wrongly calls the printer a honeypot.
                \item \textbf{gpt-oss:120b} extracts creds in $\sim$half the runs; \textbf{fabricates CVE IDs}.
                \item \textbf{qwen3:30b} discovers hosts/services but rates LDAP ``Low'' risk in 9/10 runs -- \textbf{misses the critical finding}.
                \item \textbf{Static} never performs the anonymous bind -- credentials stay unextracted despite full port coverage.
            \end{itemize}
        \end{column}
        \begin{column}{0.46\textwidth}
            \vspace{0.6em}
            \footnotesize
            \begin{itemize}
                \item Only the \textbf{frontier model} reaches the critical cleartext credentials.
                \item \textbf{SNMP} probed by nobody (only TCP sweeps); DMZ host outside the target subnet -- unreached.
            \end{itemize}

        \end{column}
    \end{columns}

    \vspace{0.3em}
    \footnotesize\textbf{Trade-off:} static = breadth \& determinism; AI = \textbf{interpretive depth} (severity-rated assessment, next steps).
\end{frame}
