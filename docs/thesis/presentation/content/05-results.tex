\begin{frame}{Single-Agent $\rightarrow$ Multi-Agent}
    \framesubtitle{Why we decompose the task}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \uncover<1->{
                \textbf{\faExclamationTriangle~~Single agent}
                \begin{itemize}
                    \item \scriptsize \texttt{claude-opus-4-7} tight (\textasciitilde53k tok, 122\,s)
                    \item \scriptsize \texttt{qwen3:30b} scatters wide (86k tok, 362\,s)
                    \item \scriptsize \texttt{gpt-oss:120b} unreliable (963\,s)
                \end{itemize}
            }
            \vspace{0.4em}
            \uncover<2->{
                \begin{block}{\footnotesize Structured vs.\ Unstructured}
                    \scriptsize
                    \textbf{Structured} output enables automated metrics but has a
                    \textbf{high failure rate} for small models.
                    \textbf{Unstructured} was required in the BFH lab, at the cost of manual evaluation.
                \end{block}
            }
        \end{column}
    \end{columns}

    \uncover<4->{
        \vspace{0.3em}
        \footnotesize \faProjectDiagram~~Decomposition gives each phase a small, fresh context $\rightarrow$ rescues the smaller models.
    }
\end{frame}

\begin{frame}{Qualitative Results + Interactive Mode}
    \framesubtitle{The discriminator: extracting the LDAP cleartext credentials}


    \uncover<1->{
        \textbf{\faSearch~~Qualitative}
        \scriptsize
        \begin{itemize}
            \item \textbf{claude-opus-4-7} extracts the credentials in \textbf{every run} + correct attack-chain reasoning. Wrongly calls the printer a honeypot.
            \item \textbf{gpt-oss:120b} extracts creds in $\sim$half the runs; \textbf{fabricates CVE IDs}.
            \item \textbf{qwen3:30b} discovers hosts/services but rates LDAP ``Low'' in 9/10 runs -- \textbf{misses it}.
            \item \textbf{Static} never performs the anonymous bind -- creds stay unextracted.
        \end{itemize}
    }
    \uncover<3->{
        \vspace{0.3em}
        \footnotesize\textbf{Trade-off:} static = breadth \& determinism; AI = \textbf{interpretive depth} (severity-rated assessment, next steps).
    }
\end{frame}
