\begin{frame}{The Challenge}
  \quad{{\Large\faExclamationTriangle}~~~\large\textbf{Rising Cyber Threats}}\\[1em]
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item[\faAngleRight] Attackers already leverage AI~\textthreequartersemdash~defenders must keep pace
    \item[\faAngleRight] Red \& blue team exercises demand deep expertise
    \item[\faAngleRight] High recurring effort~\textthreequartersemdash~inaccessible for small and medium-sized enterprises
    \item[\faAngleRight] Scripted automation is brittle and inflexible
  \end{itemize}
\end{frame}

\begin{frame}{Hypothesis}
  \small{Integrating \textbf{agentic AI} into NSAK improves \textbf{adaptability, flexibility} and
  \textbf{automation} over static scenarios, while \textbf{lowering the operator's required expertise}
  to interpret results and plan follow-up steps.}\\
  ~\\
  ~\\
  \quad{{\Large\faFlask}~~~\large\textbf{Research Questions}}\\[1em]
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item[\faAngleRight] Can an AI agent match a scripted scenario at automated network reconnaissance?
    \item[\faAngleRight] How effectively can AI assist a human operator during a live security investigation?
    \item[\faAngleRight] Does it hold up in real-world networks beyond a controlled lab?
  \end{itemize}
\end{frame}

\begin{frame}{Our Approach}
  \quad{{\Large\faRobot}~~~\large\textbf{Agentic AI}}\\[1em]

    \item[\faAngleRight] Equip it with tool calling for autonomous execution
    \item[\faAngleRight] Benchmark against a scripted reference scenario
  \end{itemize}
  \vspace{1em}
  \begin{center}
